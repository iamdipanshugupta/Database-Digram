title Smart Elevator Control

Buildings[icon:building,color:Red]{
  building_id serial pk,
  name varchar(100) not null,
  location text,
  total_floors int not null,
  created_at timestamp,
  updated_at timestamp
}

Floors[color:blue]{
  floor_id serial pk,
  building_id int fk,
  floor_number int not null,
  created_at timestamp,
  updated_at timestamp
}
Elevetor_shaft[color:Green]{
  shaft_id serial pk,
  building_id int fk,
  start_floor int not null,
  end_floor int not null,
  shaft_label varchar(50),
  is_active boolean default true,
  created_at timestamp,
  updated_at timestamp
}

Elevators[color:yellow]{
  elevator_id serial pk,
  shaft_id int fk,
  model_number varchar(50),
  capacity int,
  created_at timestamp,
  updated_at timestamp
}
Elevator_Floor_Map {
  id serial pk,
  elevator_id int fk,
  floor_id int fk
}
floor_request[color:purple]{
   request_id serial pk,
  floor_id int fk,
  direction enum('up','down'),
  request_type enum('internal','external'),
  priority enum('normal','vip','emergency') default 'normal',
  request_status enum('pending','assigned','fulfilled','cancelled'),
  created_at timestamp,
  assigned_at timestamp,
  completed_at timestamp
}
Ride_Assignments[color: Yellow] {
  assignment_id serial pk,
  request_id int fk,
  elevator_id int fk,
  assignment_status enum('active','completed','reassigned','failed'),
  estimated_arrival int,
  created_at timestamp,
  updated_at timestamp
}
Ride_Logs[color: Blue] {
   ride_log_id serial pk,
  request_id int fk,
  elevator_id int fk,
  start_floor int,
  end_floor int,
  passenger_count int,
  started_at timestamp,
  completed_at timestamp,
  created_at timestamp
}
Elevator_Status_Tracking[color: Green] {
  status_id serial pk,
  elevator_id int fk,
  current_floor int,
  status enum('idle','moving','stopped','maintenance'),
  created_at timestamp
}
Maintenance_Schedules[color: Orange] {
schedule_id serial pk,
  elevator_id int fk,
  trigger_type enum('time_based','usage_based','sensor_threshold'),
  threshold_value int,
  last_maintenance_at timestamp,
  next_due_at timestamp,
  created_at timestamp,
  updated_at timestamp
}
Maintenance_Logs[color: Purple] {
   maintenance_log_id serial pk,
  schedule_id int fk,
  issue_category enum('motor','cables','doors','software','sensors'),
  action_taken text,
  downtime_duration interval,
  created_at timestamp
}

Buildings.building_id  <  Floors.building_id
Buildings.building_id  <  Elevator_Shafts.building_id

Elevator_Shafts.shaft_id  <  Elevators.shaft_id

Floors.floor_id  <  Floor_Requests.floor_id

Floor_Requests.request_id  <  Ride_Assignments.request_id
Elevators.elevator_id      <  Ride_Assignments.elevator_id

Elevators.elevator_id  <  Ride_Logs.elevator_id
Floor_Requests.request_id < Ride_Logs.request_id

Elevators.elevator_id  <  Elevator_Status_Tracking.elevator_id

Elevators.elevator_id  <  Maintenance_Schedules.elevator_id
Maintenance_Schedules.schedule_id  <  Maintenance_Logs.schedule_id

Elevators.elevator_id  <  Elevator_Floor_Map.elevator_id
Floors.floor_id        <  Elevator_Floor_Map.floor_id
